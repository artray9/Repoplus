// КОНФИГ АНАЛИТИКИ
// ─────────────────────────────────────────────────────────────────
var ANALYTICS = {
  TIMEZONE:          'GMT+5',
  LOOKBACK_DAYS:     7,      // дней для расчёта среднего CPL
  CPL_THRESHOLD:     50,     // % роста CPL — аномалия
  CTR_DROP_THRESHOLD:30,     // % падения CTR — алерт
  MIN_SPEND:         5,      // мин. расход чтобы проверять (фильтр шума)
};

// Индексы колонок выгрузок (0-based)
var FB_COL  = { DAY:0, CLIENT:1, CAMP:2, IMP:3, RTYPE:4, RES:5, SPEND:6, MSG:7, LPV:8, LEADS:9, IG:10, WEB:11, LCLICKS:12, CTR:13, CPM:14 };
var GG_COL  = { DAY:0, CLIENT:1, CAMP:2, IMP:3, CLICKS:4, SPEND:5, CONV:6, CTR:7, CPM:8 };
var TT_COL  = { DAY:0, CLIENT:1, CAMP:2, IMP:3, CLICKS:4, SPEND:5, CONV:6, CTR:7, CPM:8 };

// ─────────────────────────────────────────────────────────────────
// УТИЛИТЫ
// ─────────────────────────────────────────────────────────────────

function getClients_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var s  = ss.getSheetByName('Settings');
  if (!s) return [];
  var last = s.getLastRow();
  if (last < 3) return [];
  return s.getRange(3, 1, last - 2, 19).getValues().filter(function(r) {
    return r[0] && r[1]; // active + clientId
  });
}

function getSheetData_(name) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name);
  if (!sheet) return [];
  var vals = sheet.getDataRange().getValues();
  return vals.slice(1); // без заголовка
}

function fmtDate_(d) {
  if (d instanceof Date) return Utilities.formatDate(d, ANALYTICS.TIMEZONE, 'yyyy-MM-dd');
  return String(d).trim();
}

function dateStr_(daysAgo) {
  var d = new Date();
  d.setDate(d.getDate() - daysAgo);
  return Utilities.formatDate(d, ANALYTICS.TIMEZONE, 'yyyy-MM-dd');
}

function parseNum_(v) {
  if (v === null || v === undefined || v === '') return 0;
  return parseFloat(String(v).replace(',', '.')) || 0;
}

// Красивый знак изменения: +12.3% или -5.1%
function fmtDiff_(val, base) {
  if (!base || base === 0) return '—';
  var pct = Math.round((val - base) / base * 100);
  return (pct > 0 ? '📈 +' : pct < 0 ? '📉 ' : '➡️ ') + pct + '%';
}

// Определяет является ли FB кампания лидовой по Result Type
function isLeadCampaign_(rtype) {
  var rt = String(rtype).toLowerCase();
  return rt.indexOf('lead') !== -1 || rt.indexOf('messag') !== -1 ||
         rt.indexOf('contact') !== -1 || rt.indexOf('custom') !== -1;
}

// Агрегация строк выгрузки за диапазон дат для одного клиента
// Для FB: лиды берутся только из лидовых кампаний (по Result Type)
//         расход и CTR/CPM — из всех кампаний
function aggregate_(rows, clientId, col, dateFrom, dateTo) {
  var result = { spend: 0, leadSpend: 0, leads: 0, clicks: 0, imp: 0, conv: 0, ctrSum: 0, cpmSum: 0, count: 0 };
  rows.forEach(function(r) {
    if (String(r[col.CLIENT]).trim() !== clientId) return;
    var d = fmtDate_(r[col.DAY]);
    if (d < dateFrom || d > dateTo) return;

    var rowSpend = parseNum_(r[col.SPEND]);
    result.spend += rowSpend;
    result.imp   += parseNum_(r[col.IMP]);
    result.count++;

    if (col === FB_COL) {
      // Лиды и расход лидовых кампаний — отдельно для корректного CPL
      if (isLeadCampaign_(r[col.RTYPE])) {
        result.leads     += parseNum_(r[col.RES]);
        result.leadSpend += rowSpend;
      }
      result.clicks += parseNum_(r[col.LCLICKS]);
      result.ctrSum += parseNum_(r[col.CTR]);
      result.cpmSum += parseNum_(r[col.CPM]);
    } else {
      result.clicks += parseNum_(r[col.CLICKS]);
      result.conv   += parseNum_(r[col.CONV]);
      result.ctrSum += parseNum_(r[col.CTR]);
      result.cpmSum += parseNum_(r[col.CPM]);
    }
  });
  result.avgCtr = result.count > 0 ? Math.round(result.ctrSum / result.count * 100) / 100 : 0;
  result.avgCpm = result.count > 0 ? Math.round(result.cpmSum / result.count * 100) / 100 : 0;
  // CPL считаем только от расхода лидовых кампаний
  result.cpl    = result.leads > 0 ? Math.round(result.leadSpend / result.leads * 100) / 100 : 0;
  return result;
}

// ─────────────────────────────────────────────────────────────────
// 1. АНОМАЛИИ CPL
// ─────────────────────────────────────────────────────────────────
function checkCplAnomalies(fbData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  var yest    = dateStr_(1);
  var histFrom = dateStr_(ANALYTICS.LOOKBACK_DAYS + 1);
  var histTo   = dateStr_(2); // до позавчера (не включаем вчера в базу)

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    // Только лидовые кампании
    var leadRows = fbData.filter(function(r) {
      if (String(r[FB_COL.CLIENT]).trim() !== clientId) return false;
      var rt = String(r[FB_COL.RTYPE]).toLowerCase();
      return rt.indexOf('lead') !== -1 || rt.indexOf('messag') !== -1 ||
             rt.indexOf('contact') !== -1 || rt.indexOf('custom') !== -1;
    });

    // Вчера
    var yA = { spend: 0, leads: 0 };
    leadRows.forEach(function(r) {
      if (fmtDate_(r[FB_COL.DAY]) !== yest) return;
      yA.spend += parseNum_(r[FB_COL.SPEND]);
      yA.leads += parseNum_(r[FB_COL.RES]);  // Results — без двойного счёта
    });

    if (yA.spend < ANALYTICS.MIN_SPEND) return;
    var yestCpl = yA.leads > 0 ? yA.spend / yA.leads : yA.spend;

    // История (база)
    var hA = { spend: 0, leads: 0 };
    leadRows.forEach(function(r) {
      var d = fmtDate_(r[FB_COL.DAY]);
      if (d < histFrom || d > histTo) return;
      hA.spend += parseNum_(r[FB_COL.SPEND]);
      hA.leads += parseNum_(r[FB_COL.RES]);
    });

    if (hA.spend === 0) return;
    var avgCpl   = hA.leads > 0 ? hA.spend / hA.leads : hA.spend;
    var diffPct  = Math.round((yestCpl - avgCpl) / avgCpl * 100);

    var msg = '';
    if (yA.leads === 0) {
      msg = '🚨 <b>0 лидов при расходе $' + yA.spend.toFixed(2) + '</b>\nПроверьте кампании — возможно сломалась посадочная или отклонили объявление.';
    } else if (diffPct >= ANALYTICS.CPL_THRESHOLD) {
      msg = '⚠️ <b>CPL вырос на ' + diffPct + '%</b>\n'
          + 'Вчера: <b>$' + yestCpl.toFixed(2) + '</b> за лид\n'
          + 'Среднее за ' + ANALYTICS.LOOKBACK_DAYS + ' дн: $' + avgCpl.toFixed(2) + '\n'
          + 'Лидов вчера: ' + yA.leads + ' / Расход: $' + yA.spend.toFixed(2);
    }

    if (msg) {
      sendToTelegram(chatId, topicId,
        '📡 <b>Мониторинг — ' + reportName + '</b>\n' + msg +
        '\n\n<i>Проверьте связки и объявления</i>'
      );
    }
  });
}

// ─────────────────────────────────────────────────────────────────
// 2. ВЧЕРА vs ПОЗАВЧЕРА
// ─────────────────────────────────────────────────────────────────
function dailyComparison(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');

  var yest = dateStr_(1);
  var dayB = dateStr_(2);

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    var fbY = aggregate_(fbData, clientId, FB_COL, yest, yest);
    var fbB = aggregate_(fbData, clientId, FB_COL, dayB, dayB);
    var ggY = aggregate_(ggData, clientId, GG_COL, yest, yest);
    var ggB = aggregate_(ggData, clientId, GG_COL, dayB, dayB);
    var ttY = aggregate_(ttData, clientId, TT_COL, yest, yest);
    var ttB = aggregate_(ttData, clientId, TT_COL, dayB, dayB);

    // Пропускаем если нет данных вообще
    if (fbY.spend + ggY.spend + ttY.spend === 0) return;

    var d1 = formatShortDate(yest);
    var d2 = formatShortDate(dayB);

    var msg = '📊 <b>День к дню — ' + reportName + '</b>\n'
            + d2 + ' → ' + d1 + '\n';

    // FB
    if (fbY.spend > 0 || fbB.spend > 0) {
      msg += '\n<b>Instagram / FB:</b>\n'
           + 'Лиды: ' + fbB.leads + ' → <b>' + fbY.leads + '</b> ' + fmtDiff_(fbY.leads, fbB.leads) + '\n'
           + 'Расход: $' + fbB.spend.toFixed(2) + ' → <b>$' + fbY.spend.toFixed(2) + '</b> ' + fmtDiff_(fbY.spend, fbB.spend) + '\n';
      if (fbB.cpl > 0 || fbY.cpl > 0)
        msg += 'CPL: $' + fbB.cpl.toFixed(2) + ' → <b>$' + fbY.cpl.toFixed(2) + '</b> ' + fmtDiff_(fbY.cpl, fbB.cpl) + '\n';
      if (fbB.avgCtr > 0 || fbY.avgCtr > 0)
        msg += 'CTR: ' + fbB.avgCtr + '% → <b>' + fbY.avgCtr + '%</b> ' + fmtDiff_(fbY.avgCtr, fbB.avgCtr) + '\n';
    }

    // Google
    if (ggY.spend > 0 || ggB.spend > 0) {
      msg += '\n<b>Google Ads:</b>\n'
           + 'Конверсии: ' + ggB.conv + ' → <b>' + ggY.conv + '</b> ' + fmtDiff_(ggY.conv, ggB.conv) + '\n'
           + 'Клики: ' + ggB.clicks + ' → <b>' + ggY.clicks + '</b> ' + fmtDiff_(ggY.clicks, ggB.clicks) + '\n'
           + 'Расход: $' + ggB.spend.toFixed(2) + ' → <b>$' + ggY.spend.toFixed(2) + '</b> ' + fmtDiff_(ggY.spend, ggB.spend) + '\n';
      if (ggB.avgCtr > 0 || ggY.avgCtr > 0)
        msg += 'CTR: ' + ggB.avgCtr + '% → <b>' + ggY.avgCtr + '%</b> ' + fmtDiff_(ggY.avgCtr, ggB.avgCtr) + '\n';
    }

    // TikTok
    if (ttY.spend > 0 || ttB.spend > 0) {
      msg += '\n<b>TikTok Ads:</b>\n'
           + 'Конверсии: ' + ttB.conv + ' → <b>' + ttY.conv + '</b> ' + fmtDiff_(ttY.conv, ttB.conv) + '\n'
           + 'Расход: $' + ttB.spend.toFixed(2) + ' → <b>$' + ttY.spend.toFixed(2) + '</b> ' + fmtDiff_(ttY.spend, ttB.spend) + '\n';
      if (ttB.avgCtr > 0 || ttY.avgCtr > 0)
        msg += 'CTR: ' + ttB.avgCtr + '% → <b>' + ttY.avgCtr + '%</b> ' + fmtDiff_(ttY.avgCtr, ttB.avgCtr) + '\n';
    }

    sendToTelegram(chatId, topicId, msg);
    Utilities.sleep(300);
  });
}

// 3. ЭТА НЕДЕЛЯ vs ПРОШЛАЯ
// ─────────────────────────────────────────────────────────────────
function weeklyComparison(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');

  // Эта неделя: -7 дней до вчера
  var thisFrom = dateStr_(7);
  var thisTo   = dateStr_(1);
  // Прошлая неделя: -14 дней до -8 дней
  var prevFrom = dateStr_(14);
  var prevTo   = dateStr_(8);

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    var fbT = aggregate_(fbData, clientId, FB_COL, thisFrom, thisTo);
    var fbP = aggregate_(fbData, clientId, FB_COL, prevFrom, prevTo);
    var ggT = aggregate_(ggData, clientId, GG_COL, thisFrom, thisTo);
    var ggP = aggregate_(ggData, clientId, GG_COL, prevFrom, prevTo);
    var ttT = aggregate_(ttData, clientId, TT_COL, thisFrom, thisTo);
    var ttP = aggregate_(ttData, clientId, TT_COL, prevFrom, prevTo);

    if (fbT.spend + ggT.spend + ttT.spend === 0) return;

    var msg = '📅 <b>Неделя к неделе — ' + reportName + '</b>\n'
            + formatShortDate(prevFrom) + '–' + formatShortDate(prevTo)
            + ' → ' + formatShortDate(thisFrom) + '–' + formatShortDate(thisTo) + '\n';

    // FB
    if (fbT.spend > 0 || fbP.spend > 0) {
      msg += '\n<b>Instagram / FB:</b>\n'
           + 'Лиды: ' + fbP.leads + ' → <b>' + fbT.leads + '</b> ' + fmtDiff_(fbT.leads, fbP.leads) + '\n'
           + 'Расход: $' + fbP.spend.toFixed(2) + ' → <b>$' + fbT.spend.toFixed(2) + '</b> ' + fmtDiff_(fbT.spend, fbP.spend) + '\n';
      if (fbP.cpl > 0 || fbT.cpl > 0)
        msg += 'CPL: $' + fbP.cpl.toFixed(2) + ' → <b>$' + fbT.cpl.toFixed(2) + '</b> ' + fmtDiff_(fbT.cpl, fbP.cpl) + '\n';
      msg += 'CTR: ' + fbP.avgCtr + '% → <b>' + fbT.avgCtr + '%</b> ' + fmtDiff_(fbT.avgCtr, fbP.avgCtr) + '\n'
           + 'CPM: $' + fbP.avgCpm + ' → <b>$' + fbT.avgCpm + '</b> ' + fmtDiff_(fbT.avgCpm, fbP.avgCpm) + '\n';
    }

    // Google
    if (ggT.spend > 0 || ggP.spend > 0) {
      msg += '\n<b>Google Ads:</b>\n'
           + 'Конверсии: ' + ggP.conv + ' → <b>' + ggT.conv + '</b> ' + fmtDiff_(ggT.conv, ggP.conv) + '\n'
           + 'Клики: ' + ggP.clicks + ' → <b>' + ggT.clicks + '</b> ' + fmtDiff_(ggT.clicks, ggP.clicks) + '\n'
           + 'Расход: $' + ggP.spend.toFixed(2) + ' → <b>$' + ggT.spend.toFixed(2) + '</b> ' + fmtDiff_(ggT.spend, ggP.spend) + '\n'
           + 'CTR: ' + ggP.avgCtr + '% → <b>' + ggT.avgCtr + '%</b> ' + fmtDiff_(ggT.avgCtr, ggP.avgCtr) + '\n'
           + 'CPM: $' + ggP.avgCpm + ' → <b>$' + ggT.avgCpm + '</b> ' + fmtDiff_(ggT.avgCpm, ggP.avgCpm) + '\n';
    }

    // TikTok
    if (ttT.spend > 0 || ttP.spend > 0) {
      msg += '\n<b>TikTok Ads:</b>\n'
           + 'Конверсии: ' + ttP.conv + ' → <b>' + ttT.conv + '</b> ' + fmtDiff_(ttT.conv, ttP.conv) + '\n'
           + 'Расход: $' + ttP.spend.toFixed(2) + ' → <b>$' + ttT.spend.toFixed(2) + '</b> ' + fmtDiff_(ttT.spend, ttP.spend) + '\n'
           + 'CTR: ' + ttP.avgCtr + '% → <b>' + ttT.avgCtr + '%</b> ' + fmtDiff_(ttT.avgCtr, ttP.avgCtr) + '\n'
           + 'CPM: $' + ttP.avgCpm + ' → <b>$' + ttT.avgCpm + '</b> ' + fmtDiff_(ttT.avgCpm, ttP.avgCpm) + '\n';
    }

    sendToTelegram(chatId, topicId, msg);
    Utilities.sleep(300);
  });
}

// ─────────────────────────────────────────────────────────────────
// 4. ГРАФИК CTR ПО ДНЯМ (Google Charts → отправка как фото)
// ─────────────────────────────────────────────────────────────────
function weeklyCtrChart(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');
  var token   = CONFIG.TG.BOT_TOKEN();

  // Последние 7 дней
  var days = [];
  for (var i = 7; i >= 1; i--) days.push(dateStr_(i));

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    // CTR по дням для каждой платформы
    var fbCtrs = [], ggCtrs = [], ttCtrs = [], labels = [];

    days.forEach(function(d) {
      labels.push(d.slice(5)); // MM-DD

      var fbDay = aggregate_(fbData, clientId, FB_COL, d, d);
      var ggDay = aggregate_(ggData, clientId, GG_COL, d, d);
      var ttDay = aggregate_(ttData, clientId, TT_COL, d, d);

      fbCtrs.push(fbDay.avgCtr);
      ggCtrs.push(ggDay.avgCtr);
      ttCtrs.push(ttDay.avgCtr);
    });

    // Строим URL Google Image Charts
    var hasData = fbCtrs.some(function(v){return v>0;}) ||
                  ggCtrs.some(function(v){return v>0;}) ||
                  ttCtrs.some(function(v){return v>0;});
    if (!hasData) return;

    var chartUrl = buildCtrChartUrl_(labels, fbCtrs, ggCtrs, ttCtrs, reportName);

    // Скачиваем PNG с QuickChart и отправляем как файл в Telegram
    try {
      var imgResp = UrlFetchApp.fetch(chartUrl, { muteHttpExceptions: true });
      if (imgResp.getResponseCode() !== 200) {
        Logger.log('QuickChart error ' + clientId + ': HTTP ' + imgResp.getResponseCode());
        return;
      }
      var imgBlob = imgResp.getBlob().setName('ctr_chart.png').setContentType('image/png');

      var formData = {
        chat_id: chatId.toString(),
        caption: '📈 CTR по дням (7 дней) — ' + reportName,
        parse_mode: 'HTML',
        photo: imgBlob,
      };
      if (topicId) formData.message_thread_id = topicId.toString();

      UrlFetchApp.fetch('https://api.telegram.org/bot' + token + '/sendPhoto', {
        method: 'post',
        payload: formData,
        muteHttpExceptions: true,
      });
    } catch(e) {
      Logger.log('weeklyCtrChart error ' + clientId + ': ' + e.message);
    }

    Utilities.sleep(500);
  });
}

function buildCtrChartUrl_(labels, fb, gg, tt, title) {
  // QuickChart.io — бесплатный Chart.js рендерер
  var chartConfig = {
    type: 'line',
    data: {
      labels: labels,
      datasets: [
        {
          label: 'FB/Instagram',
          data: fb,
          borderColor: '#4267B2',
          backgroundColor: 'rgba(66,103,178,0.1)',
          borderWidth: 2,
          pointRadius: 4,
          fill: false,
          tension: 0.3,
        },
        {
          label: 'Google Ads',
          data: gg,
          borderColor: '#EA4335',
          backgroundColor: 'rgba(234,67,53,0.1)',
          borderWidth: 2,
          pointRadius: 4,
          fill: false,
          tension: 0.3,
        },
        {
          label: 'TikTok',
          data: tt,
          borderColor: '#010101',
          backgroundColor: 'rgba(1,1,1,0.08)',
          borderWidth: 2,
          pointRadius: 4,
          fill: false,
          tension: 0.3,
        },
      ],
    },
    options: {
      title: {
        display: true,
        text: 'CTR % — ' + title,
        fontSize: 14,
      },
      legend: { position: 'bottom' },
      scales: {
        yAxes: [{
          ticks: { beginAtZero: true },
          scaleLabel: { display: true, labelString: 'CTR %' },
        }],
        xAxes: [{
          scaleLabel: { display: true, labelString: 'Дата' },
        }],
      },
    },
  };

  var url = 'https://quickchart.io/chart'
    + '?w=600&h=300'
    + '&bkg=white'
    + '&c=' + encodeURIComponent(JSON.stringify(chartConfig));

  return url;
}

// ─────────────────────────────────────────────────────────────────
// 5. АЛЕРТ: КАМПАНИЯ ОСТАНОВИЛАСЬ
// ─────────────────────────────────────────────────────────────────
function checkCampaignStopped(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');

  var yest = dateStr_(1);
  var dayB = dateStr_(2); // позавчера как база — был расход

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    var alerts = [];

    // Для каждой платформы: если позавчера был расход, а вчера 0
    var checks = [
      { data: fbData,  col: FB_COL,  name: 'Instagram/FB' },
      { data: ggData,  col: GG_COL,  name: 'Google Ads'   },
      { data: ttData,  col: TT_COL,  name: 'TikTok Ads'   },
    ];

    checks.forEach(function(c) {
      var spendYest = aggregate_(c.data, clientId, c.col, yest, yest).spend;
      var spendBase = aggregate_(c.data, clientId, c.col, dayB, dayB).spend;

      if (spendBase >= ANALYTICS.MIN_SPEND && spendYest === 0) {
        alerts.push('⛔ <b>' + c.name + '</b> — вчера $0 (позавчера $' + spendBase.toFixed(2) + ')');
      }
    });

    if (alerts.length > 0) {
      sendToTelegram(chatId, topicId,
        '🚨 <b>Стоп рекламы — ' + reportName + '</b>\n\n'
        + alerts.join('\n')
        + '\n\n<i>Проверьте бюджет, статус объявлений и платёж</i>'
      );
      Utilities.sleep(300);
    }
  });
}

// 6. АЛЕРТ: ПАДЕНИЕ CTR
// ─────────────────────────────────────────────────────────────────
function checkCtrDrop(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');

  var yest     = dateStr_(1);
  var histFrom = dateStr_(ANALYTICS.LOOKBACK_DAYS + 1);
  var histTo   = dateStr_(2);

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    var alerts = [];

    var checks = [
      { data: fbData,  col: FB_COL,  name: 'Instagram/FB' },
      { data: ggData,  col: GG_COL,  name: 'Google Ads'   },
      { data: ttData,  col: TT_COL,  name: 'TikTok Ads'   },
    ];

    checks.forEach(function(c) {
      var today = aggregate_(c.data, clientId, c.col, yest, yest);
      var hist  = aggregate_(c.data, clientId, c.col, histFrom, histTo);

      if (today.spend < ANALYTICS.MIN_SPEND || hist.avgCtr === 0) return;
      var drop = Math.round((hist.avgCtr - today.avgCtr) / hist.avgCtr * 100);
      if (drop >= ANALYTICS.CTR_DROP_THRESHOLD) {
        alerts.push('📉 <b>' + c.name + '</b>\n'
          + 'CTR вчера: ' + today.avgCtr + '%\n'
          + 'Среднее ' + ANALYTICS.LOOKBACK_DAYS + ' дн: ' + hist.avgCtr + '%\n'
          + 'Падение: -' + drop + '%'
        );
      }
    });

    if (alerts.length > 0) {
      sendToTelegram(chatId, topicId,
        '⚠️ <b>Падение CTR — ' + reportName + '</b>\n\n'
        + alerts.join('\n\n')
        + '\n\n<i>Возможно выгорание аудитории или объявлений</i>'
      );
      Utilities.sleep(300);
    }
  });
}

function weeklyTopCampaigns(fbData, ggData, ttData) {
  var clients = getClients_();
  fbData = fbData || getSheetData_('Выгрузка_FB');
  ggData = ggData || getSheetData_('Выгрузка_Google');
  ttData = ttData || getSheetData_('Выгрузка_TikTok');

  var weekFrom = dateStr_(7);
  var weekTo   = dateStr_(1);

  clients.forEach(function(row) {
    var clientId   = String(row[1]).trim();
    var reportName = row[2] || clientId;
    var chatId     = row[15];
    var topicId    = row[16];
    if (!chatId) return;

    var msg = '🏆 <b>Топ кампаний за неделю — ' + reportName + '</b>\n'
            + formatShortDate(weekFrom) + ' – ' + formatShortDate(weekTo) + '\n';

    var hasAny = false;

    // FB — топ по лидам
    var fbCamps = aggregateByCampaign_(fbData, clientId, FB_COL, weekFrom, weekTo, 'leads');
    if (fbCamps.length > 0) {
      hasAny = true;
      msg += '\n<b>Instagram/FB — топ по лидам:</b>\n';
      fbCamps.slice(0, 3).forEach(function(c, i) {
        msg += (i+1) + '. ' + truncate_(c.name, 35) + '\n'
             + '   Лиды: ' + c.leads + ' / CPL: $' + c.cpl + ' / Расход: $' + c.spend.toFixed(2) + '\n';
      });
    }

    // Google — топ по конверсиям
    var ggCamps = aggregateByCampaign_(ggData, clientId, GG_COL, weekFrom, weekTo, 'conv');
    if (ggCamps.length > 0) {
      hasAny = true;
      msg += '\n<b>Google Ads — топ по конверсиям:</b>\n';
      ggCamps.slice(0, 3).forEach(function(c, i) {
        msg += (i+1) + '. ' + truncate_(c.name, 35) + '\n'
             + '   Конв: ' + c.conv + ' / Расход: $' + c.spend.toFixed(2) + '\n';
      });
    }

    // TikTok — топ по конверсиям
    var ttCamps = aggregateByCampaign_(ttData, clientId, TT_COL, weekFrom, weekTo, 'conv');
    if (ttCamps.length > 0) {
      hasAny = true;
      msg += '\n<b>TikTok Ads — топ по конверсиям:</b>\n';
      ttCamps.slice(0, 3).forEach(function(c, i) {
        msg += (i+1) + '. ' + truncate_(c.name, 35) + '\n'
             + '   Конв: ' + c.conv + ' / Расход: $' + c.spend.toFixed(2) + '\n';
      });
    }

    if (!hasAny) return;
    sendToTelegram(chatId, topicId, msg);
    Utilities.sleep(300);
  });
}

function aggregateByCampaign_(rows, clientId, col, dateFrom, dateTo, sortBy) {
  var camps = {};
  rows.forEach(function(r) {
    if (String(r[col.CLIENT]).trim() !== clientId) return;
    var d = fmtDate_(r[col.DAY]);
    if (d < dateFrom || d > dateTo) return;
    var name = String(r[col.CAMP]);
    if (!camps[name]) camps[name] = { name: name, spend: 0, leads: 0, conv: 0, clicks: 0, imp: 0 };
    camps[name].spend += parseNum_(r[col.SPEND]);
    camps[name].imp   += parseNum_(r[col.IMP]);
    if (col === FB_COL) {
      // Лиды только из лидовых кампаний
      if (isLeadCampaign_(r[col.RTYPE])) {
        camps[name].leads += parseNum_(r[col.RES]);
      }
    } else {
      camps[name].conv  += parseNum_(r[col.CONV]);
      camps[name].clicks += parseNum_(r[col.CLICKS]);
    }
  });

  return Object.values(camps)
    .map(function(c) {
      c.cpl = c.leads > 0 ? Math.round(c.spend / c.leads * 100) / 100 : 0;
      return c;
    })
    .filter(function(c) { return c.spend > 0; })
    .sort(function(a, b) { return (b[sortBy] || 0) - (a[sortBy] || 0); });
}

function truncate_(str, len) {
  return str.length > len ? str.substring(0, len) + '…' : str;
}

// 8. ПРОВЕРКА СРОКА ТОКЕНА FB
// Назначь триггер раз в день — предупредит за 5 дней до истечения
// ─────────────────────────────────────────────────────────────────
function checkFbTokenExpiry() {
  var token = CONFIG.FB.ACCESS_TOKEN();
  if (!token) { Logger.log('checkFbTokenExpiry: нет токена'); return; }

  var res;
  try {
    res = JSON.parse(UrlFetchApp.fetch(
      'https://graph.facebook.com/debug_token'
      + '?input_token=' + token
      + '&access_token=' + token,
      { muteHttpExceptions: true }
    ).getContentText());
  } catch(e) {
    Logger.log('checkFbTokenExpiry error: ' + e.message);
    return;
  }

  if (!res.data) { Logger.log('debug_token: ' + JSON.stringify(res)); return; }

  var data     = res.data;
  var isValid  = data.is_valid;
  var expiresAt = data.expires_at ? new Date(data.expires_at * 1000) : null;
  var daysLeft  = expiresAt ? Math.round((expiresAt - new Date()) / (1000*60*60*24)) : null;

  Logger.log('FB токен: ' + (isValid ? 'действителен' : 'НЕДЕЙСТВИТЕЛЕН')
    + (expiresAt ? ', истекает ' + Utilities.formatDate(expiresAt, ANALYTICS.TIMEZONE, 'dd.MM.yyyy')
                   + ' (через ' + daysLeft + ' дн.)' : ', бессрочный'));

  // Уведомляем всех активных клиентов с chatId (или можно захардкодить один чат)
  var clients = getClients_();
  var notified = false;

  if (!isValid) {
    // Токен уже недействителен — срочно
    clients.forEach(function(row) {
      if (!row[15] || notified) return;
      sendToTelegram(row[15], row[16],
        '🔴 <b>FB токен НЕДЕЙСТВИТЕЛЕН!</b>\n'
        + 'Выгрузка Facebook не работает.\n'
        + 'Немедленно обнови FB_ACCESS_TOKEN в Script Properties.'
      );
      notified = true; // отправляем только один раз (первому клиенту с chatId)
    });
  } else if (daysLeft !== null && daysLeft <= 5) {
    clients.forEach(function(row) {
      if (!row[15] || notified) return;
      sendToTelegram(row[15], row[16],
        '⚠️ <b>FB токен истекает через ' + daysLeft + ' дн.</b>\n'
        + 'Дата: ' + Utilities.formatDate(expiresAt, ANALYTICS.TIMEZONE, 'dd.MM.yyyy') + '\n'
        + 'Обнови FB_ACCESS_TOKEN в Script Properties.'
      );
      notified = true;
    });
  }
}
